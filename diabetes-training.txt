                    tb.innerHTML = '<tr><td colspan="4" class="px-8 py-20 text-center text-gray-400 font-medium">아직 등록된 이수 데이터가 없습니다.</td></tr>';
                    return;
                }

                tb.innerHTML = '';
                let totalS = 0;
                data.forEach(item => {
                    const time = item.timestamp ? new Date(item.timestamp.seconds * 1000).toLocaleString('ko-KR') : '-';
                    totalS += item.score;
                    const row = document.createElement('tr');
                    row.className = "hover:bg-blue-50/30 transition-colors";
                    row.innerHTML = `
                        <td class="px-8 py-5 text-xs text-gray-400 font-medium">${time}</td>
                        <td class="px-8 py-5">
                            <div class="text-sm font-bold text-gray-800">${item.dept}</div>
                            <div class="text-[10px] text-gray-500 uppercase tracking-tighter">${item.team}</div>
                        </td>
                        <td class="px-8 py-5">
                            <div class="text-sm font-bold text-gray-700">${item.name}</div>
                            <div class="text-[10px] text-gray-400 font-mono">${item.empId}</div>
                        </td>
                        <td class="px-8 py-5 text-center">
                            <span class="px-3 py-1 bg-blue-50 text-blue-700 rounded-lg font-black text-sm">${item.score}</span>
                        </td>
                    `;
                    tb.appendChild(row);
                });

                document.getElementById('stat-total').innerText = data.length;
                document.getElementById('stat-avg').innerText = (totalS / data.length).toFixed(1);
                document.getElementById('stat-time').innerText = new Date().toLocaleTimeString();

            } catch (e) {
                tb.innerHTML = `<tr><td colspan="4" class="px-8 py-20 text-center text-red-400">오류 발생: ${e.message}</td></tr>`;
            }
        };

        window.exportToExcel = () => {
            if (state.fetchedData.length === 0) return;
            
            const headers = ["응시일시", "사업부", "팀", "사번", "성명", "점수"];
            const csvRows = [
                headers.join(","),
                ...state.fetchedData.map(d => [
                    d.timestamp ? `"${new Date(d.timestamp.seconds * 1000).toLocaleString('ko-KR')}"` : '"-"',
                    `"${d.dept}"`,
                    `"${d.team}"`,
                    `"${d.empId}"`,
                    `"${d.name}"`,
                    d.score
                ].join(","))
            ].join("\n");

            // BOM for Korean text in Excel
            const blob = new Blob(["\ufeff" + csvRows], { type: 'text/csv;charset=utf-8;' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.setAttribute('href', url);
            a.setAttribute('download', `당뇨교육_이수명단_${new Date().toISOString().split('T')[0]}.csv`);
            a.click();
        };

        init();
    </script>
